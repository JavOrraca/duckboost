DuckDB community extension

Train trees. Ship SQL.

Fit gradient-boosted trees inside DuckDB. Export the model as a query you can run anywhere DuckDB runs.

Get started

›_ SQL workbench

DuckDB

Install

-- Start a DuckDB session
INSTALL duckboost FROM community;
LOAD duckboost;

Train

CREATE TABLE fit AS
SELECT duckboost_train(y, [x]) AS model
FROM (SELECT i::DOUBLE AS x,
             (i * i)::DOUBLE AS y
      FROM range(100) t(i)) data;

Export

CREATE TABLE scoring_rows AS
SELECT 4.0::DOUBLE AS x;
SELECT duckboost_to_sql(
  model, 'scoring_rows', ['x'])
FROM fit;
